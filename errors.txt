// src/utils/errors.ts
import { Request, Response, NextFunction } from 'express';

// ─── Typed API error ───────────────────────────────────────────────────────
export class ApiError extends Error {
  readonly statusCode: number;
  readonly code: string;
  readonly details?: unknown;

  constructor(
    statusCode: number,
    message: string,
    code = 'API_ERROR',
    details?: unknown
  ) {
    super(message);
    this.name = 'ApiError';
    this.statusCode = statusCode;
    this.code = code;
    this.details = details;
    Error.captureStackTrace(this, ApiError);
  }
}

// ─── Async handler — eliminates try/catch boilerplate in routes ────────────
export function asyncHandler(
  fn: (req: Request, res: Response, next: NextFunction) => Promise<unknown>
) {
  return (req: Request, res: Response, next: NextFunction) => {
    Promise.resolve(fn(req, res, next)).catch(next);
  };
}

// ─── Common error factories ────────────────────────────────────────────────
export const Errors = {
  notFound:       (msg = 'Resource not found')  => new ApiError(404, msg, 'NOT_FOUND'),
  unauthorized:   (msg = 'Authentication required') => new ApiError(401, msg, 'UNAUTHORIZED'),
  forbidden:      (msg = 'Access denied')        => new ApiError(403, msg, 'FORBIDDEN'),
  badRequest:     (msg: string, details?: unknown) => new ApiError(400, msg, 'BAD_REQUEST', details),
  conflict:       (msg: string)                  => new ApiError(409, msg, 'CONFLICT'),
  tooManyRequests:(msg = 'Too many requests')    => new ApiError(429, msg, 'RATE_LIMITED'),
  internal:       (msg = 'Internal server error') => new ApiError(500, msg, 'INTERNAL'),
  serviceUnavail: (msg = 'Service unavailable')  => new ApiError(503, msg, 'SERVICE_UNAVAILABLE'),
};
